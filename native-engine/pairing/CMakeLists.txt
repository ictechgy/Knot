cmake_minimum_required(VERSION 3.22)
project(KnotPairing LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" upper_config)
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${upper_config} "${CMAKE_BINARY_DIR}/lib")
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${upper_config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
set(OPENSSL_NO_ASM ON CACHE BOOL "No assembly tool dependency" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Pinned crypto remains static" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build only explicitly selected synthetic checks" FORCE)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_subdirectory(vendor/boringssl upstream EXCLUDE_FROM_ALL)
if(WIN32)
  add_library(KnotNativePairing SHARED knot_pairing.cpp)
  target_compile_definitions(KnotNativePairing PRIVATE KN_PAIRING_EXPORTS)
else()
  add_library(KnotNativePairing STATIC knot_pairing.cpp)
endif()
target_include_directories(KnotNativePairing PUBLIC include)
target_link_libraries(KnotNativePairing PRIVATE crypto)
set_target_properties(KnotNativePairing PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
add_executable(KnotPairingCryptoChecks tests/crypto_checks.cpp)
target_link_libraries(KnotPairingCryptoChecks PRIVATE KnotNativePairing crypto)
find_package(Threads REQUIRED)
set(GTEST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/boringssl/third_party/googletest/googletest")
add_executable(KnotPairingUpstreamSpakeChecks
  vendor/boringssl/crypto/curve25519/spake25519_test.cc
  "${GTEST_ROOT}/src/gtest-all.cc" tests/upstream_main.cpp)
target_include_directories(KnotPairingUpstreamSpakeChecks PRIVATE "${GTEST_ROOT}" "${GTEST_ROOT}/include")
target_compile_definitions(KnotPairingUpstreamSpakeChecks PRIVATE OPENSSL_NO_ASM)
target_link_libraries(KnotPairingUpstreamSpakeChecks PRIVATE crypto Threads::Threads)
